# Version information and icon for a shipped binary (resources/version.rc.in).
function(hidfido_resources target description)
  get_target_property(type ${target} TYPE)
  if(type STREQUAL "SHARED_LIBRARY")
    set(HIDFIDO_FILE_TYPE VFT_DLL)
    set(HIDFIDO_FILE_NAME "${target}.dll")
  else()
    set(HIDFIDO_FILE_TYPE VFT_APP)
    set(HIDFIDO_FILE_NAME "${target}.exe")
  endif()
  set(HIDFIDO_TARGET "${target}")
  set(HIDFIDO_FILE_DESCRIPTION "${description}")
  set(HIDFIDO_ICON_LINE "1 ICON \"${PROJECT_SOURCE_DIR}/resources/hid-fido2.ico\"")
  configure_file(${PROJECT_SOURCE_DIR}/resources/version.rc.in
                 ${CMAKE_CURRENT_BINARY_DIR}/resources/${target}.rc @ONLY)
  target_sources(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/resources/${target}.rc)
endfunction()
# Visual styles and per-monitor DPI awareness for windowed tools.
function(hidfido_manifest target)
  target_sources(${target} PRIVATE ${PROJECT_SOURCE_DIR}/resources/app.manifest)
endfunction()
